// ===== Tahun otomatis di footer =====
document.getElementById('year').textContent = new Date().getFullYear();

// ===== Jika gambar belum ada, tampilkan kotak penampung =====
document.querySelectorAll('.img-slot img').forEach(img => {
  const mark = () => img.parentElement.classList.add('noimg');
  img.addEventListener('error', mark);
  if (img.complete && img.naturalWidth === 0) mark();
});

// ===== Animasi muncul saat scroll + isi power bar =====
const io = new IntersectionObserver(entries => {
  entries.forEach(e => {
    if (!e.isIntersecting) return;
    e.target.classList.add('show');
    const fill = e.target.querySelector('.track i');
    if (fill) fill.style.width = fill.dataset.value + '%';
    io.unobserve(e.target);
  });
}, { threshold: 0.15 });
document.querySelectorAll('.reveal').forEach(el => io.observe(el));

// ===== Efek miring (tilt) pada kartu saat mouse bergerak =====
document.querySelectorAll('.tilt').forEach(el => {
  el.addEventListener('mousemove', ev => {
    const r = el.getBoundingClientRect();
    const x = (ev.clientX - r.left) / r.width - 0.5;
    const y = (ev.clientY - r.top) / r.height - 0.5;
    el.style.transform = `perspective(600px) rotateY(${x * 8}deg) rotateX(${-y * 8}deg) translateY(-4px)`;
  });
  el.addEventListener('mouseleave', () => { el.style.transform = ''; });
});

// ===== Percikan api di hero (canvas) =====
(() => {
  const canvas = document.getElementById('sparks');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  let w, h, sparks = [];

  function resize() {
    w = canvas.width = canvas.offsetWidth;
    h = canvas.height = canvas.offsetHeight;
  }
  resize();
  window.addEventListener('resize', resize);

  function make() {
    return {
      x: Math.random() * w,
      y: h + 10,
      r: Math.random() * 2.2 + 0.8,
      vy: Math.random() * 1.6 + 0.6,
      vx: (Math.random() - 0.5) * 0.8,
      life: 1,
      color: Math.random() > 0.3 ? '255,107,26' : '229,50,45'
    };
  }

  function loop() {
    ctx.clearRect(0, 0, w, h);
    if (sparks.length < 70) sparks.push(make());
    sparks.forEach((s, i) => {
      s.x += s.vx;
      s.y -= s.vy;
      s.life -= 0.004;
      ctx.beginPath();
      ctx.fillStyle = `rgba(${s.color},${Math.max(s.life, 0)})`;
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fill();
      if (s.life <= 0 || s.y < -10) sparks[i] = make();
    });
    requestAnimationFrame(loop);
  }
  loop();
})();
